	if [ -r /proc/sys/kernel/random/uuid ]; then
		SAND_BOX_BOOT_ID="$(cat /proc/sys/kernel/random/uuid)"
	else
		SAND_BOX_BOOT_ID="${SAND_BOX_BOOT_STARTED_AT_MS}-$$"

}
